% resume.tex \u2014 compact pandoc template for resume.md
% optimized for 3-page technical resume, compile with LuaLaTeX
% Fully self-contained: does not source preamble.tex. Keep the callout
% style block below in sync with preamble.tex if you change one.

\documentclass[$fontsize$]{article}
\usepackage{fontspec}
\usepackage[$geometry$]{geometry}
\usepackage{hyperref}
\usepackage{xcolor}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{microtype}
\usepackage[framemethod=TikZ]{mdframed}
\usepackage{xurl}

% fonts
\setmainfont{DejaVu Serif}
\setmonofont{DejaVu Sans Mono}

% compact paragraph spacing
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.25em}

% avoid ugly page stretching
\flushbottom
\emergencystretch=2em

% section styling
\titleformat{\section}
  {\large\bfseries}
  {}
  {0em}
  {}
  [\titlerule]
\titleformat{\subsection}
  {\normalsize\bfseries}
  {}
  {0em}
  {}

\titlespacing{\section}{0pt}{0.6em}{0.3em}
\titlespacing{\subsection}{0pt}{0.4em}{0.2em}

\hypersetup{
  colorlinks=true,
  urlcolor=blue,
  linkcolor=blue,
}

% compact lists
\setlist[itemize]{
  noitemsep,
  topsep=1pt,
  parsep=0pt,
  partopsep=0pt,
  leftmargin=1.2em
}
\setlist[description]{
  style=nextline,
  leftmargin=5.5em,
  labelwidth=5em,
  itemsep=0.2em
}

% Pandoc injects \tightlist for compact lists, but it's only defined
% in the default template. We define it here so custom templates work.
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

\pagestyle{empty}

% prevent widows/orphans
\widowpenalty=10000
\clubpenalty=10000
